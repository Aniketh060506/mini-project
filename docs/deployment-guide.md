## PROJECT OVERVIEW

### What We're Building
A real-time tourist safety monitoring system with:
- **ESP32 wearable device** → sends live sensor data (GPS, heart rate, fall detection, temperature)
- **AWS IoT Core** → receives MQTT messages from device
- **DynamoDB** → stores all data
- **Lambda + API Gateway** → business logic + REST APIs
- **SNS** → sends alerts via SMS/Email
- **Blockchain** → immutable registration proof
- **Two React frontends** → Tourist portal + Authority dashboard

### Cost: $0 (AWS Free Tier)

### Hardware You Have
- ESP32 Development Board
- NEO-6M GPS Module
- MPU6050 (Accelerometer + Gyroscope)
- MAX30102 (Heart rate + SpO2 sensor)
- DS18B20 Temperature Sensor

## Step 3.5: Main ESP32 Code

**Create file: `tourist_safety.ino`**

```cpp
// tourist_safety.ino
#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>
#include <TinyGPSPlus.h>
#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>
#include <MAX30105.h>
#include <OneWire.h>
#include <DallasTemperature.h>
#include "secrets.h"

// Sensor Objects
TinyGPSPlus gps;
Adafruit_MPU6050 mpu;
MAX30105 particleSensor;
OneWire oneWire(4); // DS18B20 on GPIO 4
DallasTemperature tempSensor(&oneWire);

// WiFi and MQTT
WiFiClientSecure wifiClient;
PubSubClient mqttClient(wifiClient);

// GPS Serial (adjust pins if needed)
HardwareSerial gpsSerial(1); // RX=16, TX=17

// MQTT Topics
const char* TELEMETRY_TOPIC = "tourist/telemetry";
const char* ALERT_TOPIC = "tourist/alerts";

// Fall Detection Variables
float lastAccelMagnitude = 0;
bool fallDetected = false;
unsigned long fallDetectionTime = 0;

// Timing
unsigned long lastPublish = 0;
const long publishInterval = 5000; // 5 seconds

void setup() {
  Serial.begin(115200);
  delay(2000);
  
  Serial.println("Tourist Safety Device Starting...");
  
  // Initialize WiFi
  setupWiFi();
  
  // Initialize Sensors
  setupSensors();
  
  // Initialize MQTT
  setupMQTT();
}

void loop() {
  // Reconnect if needed
  if (!mqttClient.connected()) {
    reconnectMQTT();
  }
  mqttClient.loop();
  
  // Read GPS
  while (gpsSerial.available() > 0) {
    gps.encode(gpsSerial.read());
  }
  
  // Publish telemetry every 5 seconds
  if (millis() - lastPublish >= publishInterval) {
    readAndPublishSensors();
    lastPublish = millis();
  }
  
  delay(100);
}

void setupWiFi() {
  Serial.print("Connecting to WiFi: ");
  Serial.println(WIFI_SSID);
  
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println("\nWiFi connected!");
  Serial.print("IP: ");
  Serial.println(WiFi.localIP());
}

void setupSensors() {
  // GPS
  gpsSerial.begin(9600, SERIAL_8N1, 16, 17); // RX=16, TX=17
  Serial.println("GPS initialized");
  
  // MPU6050
  if (!mpu.begin()) {
    Serial.println("MPU6050 not found!");
    while (1) delay(10);
  }
  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setGyroRange(MPU6050_RANGE_500_DEG);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);
  Serial.println("MPU6050 initialized");
  
  // MAX30102
  if (!particleSensor.begin(Wire, I2C_SPEED_FAST)) {
    Serial.println("MAX30102 not found!");
    while (1) delay(10);
  }
  particleSensor.setup();
  particleSensor.setPulseAmplitudeRed(0x0A);
  particleSensor.setPulseAmplitudeGreen(0);
  Serial.println("MAX30102 initialized");
  
  // DS18B20
  tempSensor.begin();
  Serial.println("DS18B20 initialized");
  
  Serial.println("All sensors ready!");
}

void setupMQTT() {
  // Configure WiFiClientSecure
  wifiClient.setCACert(ROOT_CA);
  wifiClient.setCertificate(DEVICE_CERT);
  wifiClient.setPrivateKey(PRIVATE_KEY);
  
  // Configure MQTT
  mqttClient.setServer(AWS_IOT_ENDPOINT, 8883);
  mqttClient.setBufferSize(1024);
  
  Serial.println("MQTT configured");
  reconnectMQTT();
}

void reconnectMQTT() {
  while (!mqttClient.connected()) {
    Serial.print("Connecting to AWS IoT...");
    
    if (mqttClient.connect(DEVICE_ID)) {
      Serial.println("connected!");
    } else {
      Serial.print("failed, rc=");
      Serial.print(mqttClient.state());
      Serial.println(" retrying in 5 seconds");
      delay(5000);
    }
  }
}

void readAndPublishSensors() {
  // Read MPU6050
  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);
  
  // Calculate acceleration magnitude for fall detection
  float accelMagnitude = sqrt(
    pow(a.acceleration.x, 2) + 
    pow(a.acceleration.y, 2) + 
    pow(a.acceleration.z, 2)
  );
  
  // Fall Detection Logic
  // Free fall: accel < 3 m/s², then impact: accel > 20 m/s²
  if (accelMagnitude < 3.0 && lastAccelMagnitude < 3.0) {
    // Possible free fall
    fallDetectionTime = millis();
  } else if (accelMagnitude > 20.0 && 
             (millis() - fallDetectionTime) < 1000) {
    // Impact after free fall = FALL DETECTED
    fallDetected = true;
    Serial.println("⚠️ FALL DETECTED!");
  } else {
    fallDetected = false;
  }
  lastAccelMagnitude = accelMagnitude;
  
  // Read MAX30102 (Heart Rate & SpO2)
  int32_t heartRate = 0;
  int32_t spo2 = 0;
  
  if (particleSensor.available()) {
    heartRate = particleSensor.getHeartRate();
    spo2 = particleSensor.getSpO2();
    particleSensor.nextSample();
  }
  
  // Read DS18B20 Temperature
  tempSensor.requestTemperatures();
  float bodyTemp = tempSensor.getTempCByIndex(0);
  
  // GPS Data
  float latitude = gps.location.isValid() ? gps.location.lat() : 0.0;
  float longitude = gps.location.isValid() ? gps.location.lng() : 0.0;
  float altitude = gps.altitude.isValid() ? gps.altitude.meters() : 0.0;
  
  // Create JSON payload
  StaticJsonDocument<512> doc;
  doc["deviceId"] = DEVICE_ID;
  doc["timestamp"] = millis();
  doc["latitude"] = latitude;
  doc["longitude"] = longitude;
  doc["altitude"] = altitude;
  doc["heartRate"] = heartRate;
  doc["spo2"] = spo2;
  doc["temperature"] = bodyTemp;
```

```javascript
const AWS = require('aws-sdk');
const dynamodb = new AWS.DynamoDB.DocumentClient();

const ALERTS_TABLE = 'Alerts';
const PROFILES_TABLE = 'TouristProfiles';

exports.handler = async (event) => {
  console.log('Received event:', JSON.stringify(event));
  
  try {
    const data = event;
    
    // Determine alert type
    let alertType = 'UNKNOWN';
    let severity = 'INFO';

      alertType = 'FALL_DETECTED';
      severity = 'CRITICAL';
    } else if (data.heartRate < 40 || data.heartRate > 150) {
      alertType = 'ABNORMAL_HEART_RATE';
      severity = 'WARNING';

      severity = 'CRITICAL';

    let touristId = 'UNKNOWN';
    let touristName = 'Unknown Tourist';
    let emergencyContact = null;
    
    if (profileResult.Items && profileResult.Items.length > 0) {
      const profile = profileResult.Items[0];
      touristId = profile.touristId;
      touristName = profile.name;
      emergencyContact = profile.emergencyContact;
    }
    
    // Save alert to DynamoDB
    const alert = {
      alertId: alertId,
      timestamp: Date.now(),
      deviceId: data.deviceId,
      touristId: touristId,
      alertType: alertType,
      severity: severity,
      latitude: data.latitude,
      longitude: data.longitude,
      sensorData: {
        heartRate: data.heartRate,
        spo2: data.spo2,
        temperature: data.temperature,
        accelX: data.accelX,
        accelY: data.accelY,
        accelZ: data.accelZ
      },
      status: 'ACTIVE',
      acknowledgedBy: null,
      acknowledgedAt: null
    };
    
    await dynamodb.put({
      TableName: ALERTS_TABLE,
      Item: alert
    }).promise();
    
    console.log('Alert saved:', alertId);
    
    // Send SNS notification
    const message = `🚨 ${severity} ALERT: ${alertType}
Tourist: ${touristName} (${touristId})
Device: ${data.deviceId}
Location: ${data.latitude}, ${data.longitude}
Heart Rate: ${data.heartRate} bpm
SpO2: ${data.spo2}%
Temperature: ${data.temperature}°C
Time: ${new Date().toISOString()}

View dashboard for details.`;
    
    await sns.publish({
      TopicArn: SNS_TOPIC_ARN,
      Subject: `${severity}: ${alertType}`,
      Message: message
    }).promise();
    
    console.log('SNS notification sent');
    
    return {
      statusCode: 200,
      body: JSON.stringify({ alertId: alertId })
    };
    
  } catch (error) {

    return {
      statusCode: 500,
      body: JSON.stringify({ error: error.message })

};
```

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract TouristRegistry {
    struct Tourist {
        string deviceId;
        uint256 timestamp;
        bool active;
    }
    
    mapping(uint256 => Tourist) public tourists;
    uint256 public touristCount;
    address public authority;
    
    event TouristRegistered(uint256 indexed id, string deviceId, uint256 timestamp);
    event TouristDeactivated(uint256 indexed id);
    
    constructor() {
        authority = msg.sender;
    }
    
    modifier onlyAuthority() {
        require(msg.sender == authority, "Only authority can perform this action");
        _;
    }
    
    function registerTourist(string memory _deviceId) 
        public 
        onlyAuthority 
        returns (uint256) 
    {
        touristCount++;
        tourists[touristCount] = Tourist({
            deviceId: _deviceId,
            timestamp: block.timestamp,
            active: true
        });
        
        emit TouristRegistered(touristCount, _deviceId, block.timestamp);
        return touristCount;
    }
    
    function deactivateTourist(uint256 _id) public onlyAuthority {
        require(_id > 0 && _id <= touristCount, "Invalid tourist ID");
        tourists[_id].active = false;
        emit TouristDeactivated(_id);
    }
    
    function getTourist(uint256 _id) 
        public 
        view 
        returns (string memory deviceId, uint256 timestamp, bool active) 
    {
        require(_id > 0 && _id <= touristCount, "Invalid tourist ID");
        Tourist memory t = tourists[_id];
        return (t.deviceId, t.timestamp, t.active);
    }
}
```

## Step 10.3: Fix Common Issues

### Issue: ESP32 not connecting to IoT Core
**Solution:**
- Check WiFi credentials in secrets.h
- Verify IoT endpoint is correct
- Ensure certificates are properly formatted (no extra spaces)
- Check IoT policy allows iot:Connect, iot:Publish

### Issue: Data not appearing in DynamoDB
**Solution:**
- Check IoT Rules are enabled
- Verify SQL statement in IoT Rule matches topic
- Check Lambda role has DynamoDB permissions
- Look at CloudWatch Logs for errors

### Issue: SNS not sending alerts
**Solution:**
- Verify phone number has country code (+91...)
- Check SNS subscription is "Confirmed"
- For SMS, check SNS spending limit (default $1/month)
- Increase limit: SNS → Text messaging → Spending limit

### Issue: API Gateway returns 502
**Solution:**
- Check Lambda function timeout (increase to 30s)
- Look at Lambda CloudWatch Logs for errors
- Verify Lambda has correct IAM role
- Test Lambda directly before testing via API Gateway

### Issue: Frontend CORS errors
**Solution:**
- Enable CORS in API Gateway
- Redeploy API after enabling CORS
- Add headers in Lambda response:
```javascript
headers: {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': '*'
}
```

# 🔧 TROUBLESHOOTING GUIDE

## ESP32 Issues

### Serial Monitor shows "Connecting to WiFi..." forever
**Cause:** Wrong WiFi credentials or network issue  
**Fix:**
```cpp
// In secrets.h, verify:
const char* WIFI_SSID = "YourExactWiFiName"; // Case-sensitive!
const char* WIFI_PASSWORD = "YourExactPassword";
```

---

### "Connecting to AWS IoT...failed, rc=-2"
**Cause:** Certificate issue  
**Fix:**
- Verify certificates copied correctly (no line breaks in middle)
- Check for extra spaces or newlines
- Ensure BEGIN/END lines included
- Error codes:
  - rc=-2: Network error
  - rc=-4: Connection timeout
  - rc=5: Connection refused (wrong credentials)

---

### "MPU6050 not found!"
**Cause:** Wiring issue or wrong I2C address  
**Fix:**
- Check SDA → GPIO 21, SCL → GPIO 22
- Check 3.3V and GND connected
- Run I2C scanner:
```cpp
#include <Wire.h>
void setup() {
  Wire.begin();
  Serial.begin(115200);
```

## AWS Issues

### Lambda function times out
**Cause:** Insufficient timeout or cold start  
**Fix:**
- Lambda → Configuration → General → Timeout: Set 30-60 seconds
- For blockchain calls, increase to 60 seconds
- Check CloudWatch Logs for actual error

---

### DynamoDB "AccessDeniedException"
**Cause:** Lambda role missing permissions  
**Fix:**
- IAM → Roles → TouristSafetyLambdaRole
- Permissions → Add policy → AmazonDynamoDBFullAccess
- Wait 30 seconds for policy to propagate

---

### API Gateway returns "Missing Authentication Token"
**Cause:** Wrong URL or method not deployed  
**Fix:**
- Copy EXACT invoke URL from API Gateway console
- Verify resource path: /tourists not /tourist
- Redeploy API: Actions → Deploy API → prod

---

### CORS error in browser console
**Cause:** CORS not enabled or headers missing  
**Fix:**
1. API Gateway → Select API → Actions → Enable CORS
2. In Lambda, add headers:
```javascript
return {
  statusCode: 200,
  headers: {
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Headers': 'Content-Type',
    'Access-Control-Allow-Methods': 'GET,POST,OPTIONS'
  },
  body: JSON.stringify(data)
};
```
3. Redeploy API

## Frontend Issues

### React app shows "Network Error"
**Cause:** Wrong API URL or CORS  
**Fix:**
- Check .env file has correct API_BASE_URL
- No trailing slash in URL
- API Gateway CORS enabled
- Lambda returns CORS headers

---

### Cognito login fails with "User does not exist"
**Cause:** User not created or wrong user pool  
**Fix:**
- Cognito → User pools → Select pool → Users tab
- Verify user exists
- Check USER_POOL_ID and CLIENT_ID in .env match

---

### Map not displaying
**Cause:** Missing API key or network  
**Fix:**
- If using Google Maps, get API key
- If using Leaflet (recommended), no key needed
- Check browser console for errors
- Verify latitude/longitude values are valid
